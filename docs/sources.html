Add your own source

Subclass Source, decorate with @register, and import the module. Or ship a plugin through the coldtrail.sources entry-point group.

from coldtrail.models import CheckResult, Profile
from coldtrail.sources import Source, register

@register
class Example(Source):
    name = "example"
    label = "Example"
    group = "social"

    async def check(self, fetcher, target):
        status, data = await fetcher.get_json(
            f"https://api.example.com/users/{target}"
        )
        if status == 404:
            return self.missing(target)
        if status != 200:
            return self.failed(target, f"HTTP {status}")
        return self.ok(target, Profile(
            source=self.name,
            username=target,
            url=data["html_url"],
            display_name=data.get("name"),
        ))

Then record a replayable contract fixture so upstream drift fails loudly:

uv run coldtrail source verify --record -s example
uv run coldtrail source verify -s example
